% 自然宽度 79.98pt; 压到 78pt 需要 1.98pt 收缩量
\documentclass[12pt]{article}
\usepackage[margin=1cm,paperwidth=12.5cm,paperheight=7cm]{geometry}
\usepackage{xeCJK,xeCJKfntef}
\usepackage{xcolor}
\pagestyle{empty}
\parindent=0pt \hfuzz=100pt
\begin{document}
\textbf{压进 \texttt{\string\hbox\ to\ 78pt}：自然宽度 79.98pt，需压缩 1.98pt}

\medskip
\fbox{\hbox to 78pt{\CJKunderline{虚室 hello 生白}}}\ 无编组：排得进

\medskip
\fbox{\hbox to 78pt{\CJKunderline{虚室 {hello} 生白}}}\ 有编组：溢出框外

\bigskip
\textbf{手册给出的替代写法}

\medskip
\fbox{\hbox to 78pt{\CJKunderline{虚室 \color{red}hello 生白}}}\ 花括号留给参数

\medskip
\fbox{\hbox to 78pt{\CJKunderdot{虚室 {hello} 生白}}}\ 改用符号型命令
\end{document}
