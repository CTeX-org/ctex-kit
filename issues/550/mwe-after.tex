\makeindex
\newcommand\pyindex[2]{%
  \begingroup
    \xpinyinsetup{tone=none}%
    \edef\pyindexkey{#1}%
    \expandafter\index\expandafter{\pyindexkey @#2}%
  \endgroup}
\begin{document}
\section*{四个查询命令}

\begin{tabular}{@{}ll@{}}

\verb|\xpinyininitial{汉}|  & \xpinyininitial{汉}  \\
\verb|\xpinyinshengmu{中}|  & \xpinyinshengmu{中}  \\
\verb|\xpinyinyunmu{中}|    & \xpinyinyunmu{中}    \\
\end{tabular}

\section*{多音字: 带星号取全部读音}

\begin{tabular}{@{}ll@{}}
\verb|\xpinyinvalue*{行}|   & \xpinyinvalue*{行}   \\
\verb|\xpinyininitial*{行}| & \xpinyininitial*{行} \\
\end{tabular}

\section*{tone 的四种取值}

\xpinyinsetup{tone=mark}\verb|mark|: \xpinyinvalue{女}\quad
\xpinyinsetup{tone=number}\verb|number|: \xpinyinvalue{女}\quad
\xpinyinsetup{tone=none}\verb|none|: \xpinyinvalue{女}

\section*{解决 issue 550: 索引按拼音排序}

\pyindex{\xpinyinvalue{汉}}{汉字}汉字\quad
\pyindex{\xpinyinvalue{安}}{安全}安全\quad
\pyindex{\xpinyinvalue{中}}{中国}中国

写进 \texttt{.idx} 的是 \texttt{han@汉字} 一类, 经 makeindex 排出
「安全、汉字、中国」(an/han/zhong)。
