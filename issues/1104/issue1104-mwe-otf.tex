% #1104 MWE 2：OpenType 字体没有专用的 microtype 配置时，突出量按字符宽度计算。
% 修复前 xeCJK 按 1 号字形的宽度计算，数值与只加载 fontspec 时不同。
% 把 xeCJK 换成 fontspec 即得参照值。
\documentclass{article}
\usepackage{xeCJK}
\setmainfont{texgyretermes-regular.otf}
\usepackage{microtype}
% XeTeX 按字形序号保存 OpenType 字体的 \lpcode/\rpcode，码位前加 U 才按 Unicode 码位查询。
\newcommand\showpr[1]{\typeout{U+#1: lp=\the\lpcode\font U"#1, rp=\the\rpcode\font U"#1}}
\begin{document}
x\showpr{2013}\showpr{2014}\showpr{2018}\showpr{2019}\showpr{201C}\showpr{201D}\showpr{002C}
\end{document}
