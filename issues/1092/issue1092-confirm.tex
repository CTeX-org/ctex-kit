\newcommand\row[2]{%
  \setbox\Pre=\hbox{长度约为}%
  \setbox\Alone=\hbox{#2}%
  \setbox\Left=\hbox{长度约为#2}%
  \setbox\Both=\hbox{长度约为#2后接中文}%
  \setbox\Post=\hbox{后接中文}%
  \setlength\GapL{\dimexpr\wd\Left-\wd\Pre-\wd\Alone\relax}%
  \setlength\GapR{\dimexpr\wd\Both-\wd\Left-\wd\Post\relax}%
  \begin{tikzpicture}[baseline=0pt,x=1pt,y=1pt]
    \node[anchor=base west,inner sep=0pt] at (0,0) {\copy\Both};
    \ifdim\GapL<0.01pt
      \fill[red,opacity=.85] (\wd\Pre-0.6,-3) rectangle ++(1.2,13);
      \node[red,anchor=north,font=\tiny\bfseries] at (\wd\Pre,-3) {缺};
    \else
      \fill[green!60!black,opacity=.4] (\wd\Pre,-3) rectangle ++(\GapL,13);
    \fi
    \ifdim\GapR<0.01pt
      \fill[red,opacity=.85] (\wd\Left-0.6,-3) rectangle ++(1.2,13);
    \else
      \fill[green!60!black,opacity=.4] (\wd\Left,-3) rectangle ++(\GapR,13);
    \fi
    \node[anchor=west,font=\scriptsize] at (\wd\Both+6,3)
      {左 \textcolor{\ifdim\GapL<0.01pt red\else green!40!black\fi}{\the\GapL}\quad
       右 \textcolor{\ifdim\GapR<0.01pt red\else green!40!black\fi}{\the\GapR}};
    \node[anchor=east,font=\scriptsize\ttfamily,text=gray] at (-4,3) {#1};
  \end{tikzpicture}\par\smallskip}
