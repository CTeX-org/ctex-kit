  \setlength\GapR{\dimexpr\wd\Both-\wd\Left-\wd\Post\relax}%
  \begin{tikzpicture}[baseline=0pt,x=1pt,y=1pt]
    \node[anchor=base west,inner sep=0pt] at (0,0) {\copy\Both};
    \ifdim\GapL<0.01pt \ifLexp\okfalse\else\oktrue\fi \else \ifLexp\oktrue\else\okfalse\fi \fi
    \edef\Lc{\ifok green!40!black\else red\fi}%
    \ifdim\GapL<0.01pt
      \ifok\else
        \fill[red,opacity=.85] (\wd\Pre-0.6,-3) rectangle ++(1.2,13);
        \node[red,anchor=north,font=\tiny\bfseries] at (\wd\Pre,-3) {缺};
      \fi
    \else
      \fill[\ifok green!60!black\else red\fi,opacity=.4] (\wd\Pre,-3) rectangle ++(\GapL,13);
      \ifok\else\node[red,anchor=north,font=\tiny\bfseries] at (\wd\Pre+\GapL/2,-3) {多};\fi
    \fi
    \ifdim\GapR<0.01pt \ifRexp\okfalse\else\oktrue\fi \else \ifRexp\oktrue\else\okfalse\fi \fi
    \edef\Rc{\ifok green!40!black\else red\fi}%
    \ifdim\GapR<0.01pt
      \ifok\else
        \fill[red,opacity=.85] (\wd\Left-0.6,-3) rectangle ++(1.2,13);
      \fi
    \else
      \fill[\ifok green!60!black\else red\fi,opacity=.4] (\wd\Left,-3) rectangle ++(\GapR,13);
      \ifok\else\node[red,anchor=north,font=\tiny\bfseries] at (\wd\Left+\GapR/2,-3) {多};\fi
    \fi
    \node[anchor=west,font=\scriptsize] at (\wd\Both+6,3)
      {左 \textcolor{\Lc}{\the\GapL}\quad 右 \textcolor{\Rc}{\the\GapR}};
    \node[anchor=east,font=\scriptsize\ttfamily,text=gray] at (-4,3) {#1};
  \end{tikzpicture}\par\smallskip}
